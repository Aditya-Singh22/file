-- Q3(a). Create Tables for University Course Registration System
-- ==========================================================

CREATE TABLE Department7777 (
    DepartmentID INT PRIMARY KEY,
    DepartmentName VARCHAR2(50),
    Office VARCHAR2(50)
);

CREATE TABLE Faculty7777 (
    FacultyID INT PRIMARY KEY,
    FacultyName VARCHAR2(50),
    Email VARCHAR2(50),
    Qualification VARCHAR2(50),
    DepartmentID INT,
    FOREIGN KEY (DepartmentID)
    REFERENCES Department7777(DepartmentID)
);

CREATE TABLE Course7777 (
    CourseID INT PRIMARY KEY,
    CourseName VARCHAR2(50),
    Credits INT,
    Description VARCHAR2(100),
    FacultyID INT,
    FOREIGN KEY (FacultyID)
    REFERENCES Faculty7777(FacultyID)
);

CREATE TABLE Student7777 (
    StudentID INT PRIMARY KEY,
    StudentName VARCHAR2(50),
    Email VARCHAR2(50),
    Phone VARCHAR2(15),
    DateOfBirth DATE
);

CREATE TABLE Enrollment7777 (
    StudentID INT,
    CourseID INT,
    EnrollDate DATE,
    PRIMARY KEY (StudentID, CourseID),
    FOREIGN KEY (StudentID)
    REFERENCES Student7777(StudentID),
    FOREIGN KEY (CourseID)
    REFERENCES Course7777(CourseID)
);

-- ==========================================================
-- Q3(c). Display All Students Enrolled in a Particular Course
-- ==========================================================

SELECT
    s.StudentID,
    s.StudentName,
    c.CourseName
FROM Student7777 s
JOIN Enrollment7777 e
ON s.StudentID = e.StudentID
JOIN Course7777 c
ON e.CourseID = c.CourseID
WHERE c.CourseID = 201;


-- ==========================================================
-- Q3(c). Display All Courses Taught by a Particular Faculty
-- ==========================================================

SELECT
    f.FacultyName,
    c.CourseName,
    c.Credits
FROM Faculty7777 f
JOIN Course7777 c
ON f.FacultyID = c.FacultyID
WHERE f.FacultyID = 101;


-- ==========================================================
-- Q4(a). Display All Students Belonging to a Particular Department
-- ==========================================================

SELECT
    s.StudentID,
    s.StudentName,
    d.DepartmentName
FROM Student7777 s
JOIN Enrollment7777 e
ON s.StudentID = e.StudentID
JOIN Course7777 c
ON e.CourseID = c.CourseID
JOIN Faculty7777 f
ON c.FacultyID = f.FacultyID
JOIN Department7777 d
ON f.DepartmentID = d.DepartmentID
WHERE d.DepartmentName = 'Computer';


-- ==========================================================
-- Q4(b). Display Number of Students Enrolled in Each Course
-- ==========================================================

SELECT
    c.CourseID,
    c.CourseName,
    COUNT(e.StudentID) AS TotalStudents
FROM Course7777 c
JOIN Enrollment7777 e
ON c.CourseID = e.CourseID
GROUP BY c.CourseID, c.CourseName;


-- ==========================================================
-- Q4(c). Display Faculty Members and the Courses Taught by Them
-- ==========================================================

SELECT
    f.FacultyName,
    c.CourseName
FROM Faculty7777 f
JOIN Course7777 c
ON f.FacultyID = c.FacultyID;


-- ==========================================================
-- Q4(d). Display Courses Having More Than One Student Enrolled
-- ==========================================================

SELECT
    c.CourseID,
    c.CourseName,
    COUNT(e.StudentID) AS TotalStudents
FROM Course7777 c
JOIN Enrollment7777 e
ON c.CourseID = e.CourseID
GROUP BY c.CourseID, c.CourseName
HAVING COUNT(e.StudentID) > 1;
