### Q2. Determine the required sample size using the 95% confidence level and 5% margin of error.

```python
# SAMPLE SIZE DETERMINATION

N = len(data)
Z = 1.96
P = 0.5
E = 0.05

n0 = (Z**2 * P * (1 - P)) / (E**2)

sample_size = n0 / (1 + ((n0 - 1) / N))
sample_size = ceil(sample_size)

print("Required Sample Size:", sample_size)
```

### Q3. Perform the following sampling techniques.

```python
# 3. SIMPLE RANDOM SAMPLING

random_sample = data.sample(
    n=sample_size,
    random_state=42
)

# 4. SYSTEMATIC SAMPLING

interval = N // sample_size

np.random.seed(42)
start = np.random.randint(0, interval)

systematic_sample = data.iloc[
    start::interval
].head(sample_size)

# 5. STRATIFIED SAMPLING USING DEPARTMENT

stratified_parts = []

for dept, group in data.groupby("Department"):
    group_sample_size = round(
        (len(group) / N) * sample_size
    )

    selected = group.sample(
        n=min(group_sample_size, len(group)),
        random_state=42
    )

    stratified_parts.append(selected)

stratified_sample = pd.concat(
    stratified_parts,
    ignore_index=True
)

# 6. CLUSTER SAMPLING USING EDUCATION LEVEL

clusters = data["Education_Level"].dropna().unique()

np.random.seed(42)
selected_cluster = np.random.choice(clusters)

cluster_sample = data[
    data["Education_Level"] == selected_cluster
].head(sample_size)

# 7. CONVENIENCE SAMPLING

convenience_sample = data.head(sample_size)

# 8. PURPOSIVE SAMPLING

purposive_sample = data[
    data["Monthly_Salary"] > 80000
]
```

## Q3. Calculate Cronbach's Alpha for Reliability Analysis.

```python
survey_items = df.iloc[:, 3:19].apply(
    pd.to_numeric, errors="coerce"
)

def cronbach_alpha(data):
    item_variances = data.var(axis=0, ddof=1)
    total_variance = data.sum(axis=1).var(ddof=1)
    n_items = data.shape[1]

    return (n_items / (n_items - 1)) * (
        1 - item_variances.sum() / total_variance
    )

alpha = cronbach_alpha(survey_items)

print("Cronbach Alpha:", round(alpha, 3))
```
