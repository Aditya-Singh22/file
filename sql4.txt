-- Q1. Create Address Object Type
-- ==========================================================

CREATE TYPE Address AS OBJECT (
    Street   VARCHAR2(50),
    City     VARCHAR2(30),
    Pincode  VARCHAR2(10)
);


-- ==========================================================
-- Part A
-- Q2. Create Employee Table with Composite Attribute
-- ==========================================================

CREATE TABLE Employee (
    EmpID NUMBER PRIMARY KEY,
    EmpName VARCHAR2(50),
    EmpAddress Address

-- ==========================================================
-- Part B: Object Type Inheritance
-- Q1. Create Parent Object Type
-- ==========================================================

CREATE TYPE PersonType AS OBJECT (
    PersonID NUMBER,
    PersonName VARCHAR2(50)
) NOT FINAL;


-- ==========================================================
-- Part B
-- Q2. Create Parent Object Table
-- ==========================================================

CREATE TABLE Person OF PersonType (
    PRIMARY KEY (PersonID)
);


-- ==========================================================
-- Part B
-- Q3. Insert Records into Parent Table
-- ==========================================================

INSERT INTO Person
VALUES (PersonType(1, 'Rahul'));

INSERT INTO Person
VALUES (PersonType(2, 'Priya'));

INSERT INTO Person
VALUES (PersonType(3, 'Amit'));


-- ==========================================================
-- Part B
-- Q4. Query Parent Table
-- ==========================================================

SELECT * FROM Person;


-- ==========================================================
-- Part B
-- Q5. Create Child Type Using Inheritance
-- ==========================================================

CREATE TYPE EmployeeType UNDER PersonType (
    Salary NUMBER
);


-- ==========================================================
-- Part B
-- Q6. Create Child Object Table
-- ==========================================================

CREATE TABLE EmployeeChild OF EmployeeType (
    PRIMARY KEY (PersonID)
);


-- ==========================================================
-- Part B
-- Q7. Insert Records into Child Table
-- ==========================================================

INSERT INTO EmployeeChild
VALUES (EmployeeType(101, 'Rahul', 50000));

INSERT INTO EmployeeChild
VALUES (EmployeeType(102, 'Priya', 60000));


-- ==========================================================
-- Part B
-- Q8. Query Child Table
-- ==========================================================

SELECT * FROM EmployeeChild;


-- ==========================================================
-- Part B
-- Q9. Describe Child Table Structure
-- ==========================================================

DESC EmployeeChild;

-- Q1. Create a composite type named Contact containing Phone and Email
-- ==========================================================

CREATE TYPE Contact AS OBJECT (
    Phone VARCHAR2(15),
    Email VARCHAR2(50)
);


-- ==========================================================
-- Practice Questions
-- Q2. Create a Student table using Contact type
-- ==========================================================

CREATE TABLE Student_1 (
    StudentID NUMBER PRIMARY KEY,
    StudentName VARCHAR2(50),
    StudentContact Contact
);


-- ==========================================================
-- Practice Questions
-- Q3. Insert one record into Student table
-- ==========================================================

INSERT INTO Student_1
VALUES (
    1,
    'Aditya',
    Contact('9876543210', 'aditya@gmail.com')
);


-- ==========================================================
-- Practice Questions
-- Q4. Retrieve only Email field
-- ==========================================================

SELECT
    s.StudentContact.Email AS Email
FROM Student_1 s;

-- ==========================================================

CREATE TYPE StudentType AS OBJECT (
    StudentID NUMBER,
    StudentName VARCHAR2(50),
    StudentContact Contact
) NOT FINAL;

CREATE TYPE ResearchScholarType UNDER StudentType (
    ResearchArea VARCHAR2(100)
);

CREATE TABLE ResearchScholar OF ResearchScholarType (
    PRIMARY KEY (StudentID)
);
