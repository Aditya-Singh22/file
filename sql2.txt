-- ==========================================================
-- Section B: Horizontal Fragmentation
-- Q4. Create four fragmented tables
-- ==========================================================

CREATE TABLE Patient_Mumbai AS
SELECT *
FROM Patient777
WHERE Hospital='Mumbai Hospital';

CREATE TABLE Patient_Pune AS
SELECT *
FROM Patient777
WHERE Hospital='Pune Hospital';

CREATE TABLE Patient_Nashik AS
SELECT *
FROM Patient777
WHERE Hospital='Nashik Hospital';

CREATE TABLE Patient_Nagpur AS
SELECT *
FROM Patient777
WHERE Hospital='Nagpur Hospital';


-- ==========================================================
-- Section B
-- Q5. Verify each fragmented table
-- ==========================================================

-- Section B
-- Q6. Display total patients in each hospital
-- ==========================================================

SELECT COUNT(*) AS Total_Patients FROM Patient_Nagpur;


-- ==========================================================
-- Section C: Data Replication
-- Q7. Create replica table
-- ==========================================================

CREATE TABLE Patient_Replica AS
SELECT *
FROM Patient777;


-- ==========================================================
-- Section C
-- Q8. Insert a new patient and synchronize replica
-- ==========================================================

INSERT INTO Patient777
VALUES
(113,'Arjun Nair',42,'Male','Dengue','Dr Sharma','Mumbai Hospital',DATE '2026-07-15');

INSERT INTO Patient_Replica
SELECT *
FROM Patient777
WHERE Patient_ID=113;


-- ==========================================================
-- Section C
-- Q9. Update disease and synchronize replica
-- ==========================================================

UPDATE Patient777
SET Disease='Hypertension'
WHERE Patient_ID=105;

UPDATE Patient_Replica
SET Disease='Hypertension'
WHERE Patient_ID=105;


-- ==========================================================
-- Section C
-- Q10. Delete patient and synchronize replica
-- ==========================================================

DELETE FROM Patient777
WHERE Patient_ID=102;

DELETE FROM Patient_Replica
WHERE Patient_ID=102;

-- ==========================================================
-- Section D
-- Q13. Display number of patients treated by each doctor
-- ==========================================================

SELECT Doctor_Name,
       COUNT(*) AS Total_Patients
FROM
(
    SELECT Doctor_Name FROM Patient_Mumbai
    UNION ALL
    SELECT Doctor_Name FROM Patient_Pune
    UNION ALL
    SELECT Doctor_Name FROM Patient_Nashik
    UNION ALL
    SELECT Doctor_Name FROM Patient_Nagpur
)
GROUP BY Doctor_Name;
