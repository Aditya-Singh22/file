Question 2. Publish data to the Kafka topic using a Producer.
--------------------------------------------------

Write Python code to:
1. Load the API dataset provided.
2. Create a Kafka Producer.
3. Convert each record into JSON format.
4. Publish all records to the Kafka topic.
5. Verify that the messages have been sent successfully.

Code:

import json
from kafka import KafkaProducer
from kafka.errors import NoBrokersAvailable

TOPIC_NAME = "posts_topic"
BOOTSTRAP_SERVERS = "localhost:9092"
DATASET_FILE = "posts.json"

try:
    with open(DATASET_FILE, "r", encoding="utf-8") as f:
        data = json.load(f)

    print(f"Total records found in {DATASET_FILE}: {len(data)}")

    producer = KafkaProducer(
        bootstrap_servers=BOOTSTRAP_SERVERS,
        value_serializer=lambda x: json.dumps(x).encode("utf-8")
    )

    print(f"Sending records to topic: {TOPIC_NAME}\n")

    for i, record in enumerate(data, 1):
        producer.send(TOPIC_NAME, value=record)

        if i % 10 == 0 or i <= 5:
            print(f"Record {i} sent successfully")

    producer.flush()
    producer.close()

    print(f"\nTotal Records Sent : {len(data)}")
    print(f"Topic : {TOPIC_NAME}")
    print("Status : SUCCESS")

except NoBrokersAvailable:
    print("No Kafka brokers available. Is Kafka running?")

import json
from kafka import KafkaConsumer
import pandas as pd

TOPIC_NAME = "posts_topic"
BOOTSTRAP_SERVERS = "localhost:9092"

consumer = KafkaConsumer(
    TOPIC_NAME,
    bootstrap_servers=BOOTSTRAP_SERVERS,
    auto_offset_reset="earliest",
    enable_auto_commit=True,
    value_deserializer=lambda x: json.loads(x.decode("utf-8")),
    consumer_timeout_ms=10000
)

messages = []

print(f"Consuming messages from topic: {TOPIC_NAME}\n")

for message in consumer:
    messages.append(message.value)

    if len(messages) % 10 == 0 or len(messages) <= 5:
        print(f"Message {len(messages)} received")

consumer.close()

print(f"\nTotal messages received: {len(messages)}")

df = pd.DataFrame(messages)

print(f"DataFrame shape: {df.shape}")
print("Columns:", list(df.columns))

def search_catalog(catalog, column_name, search_value):
    if column_name not in catalog.columns:
        raise ValueError(
            f"Column '{column_name}' does not exist in the catalog."
        )

    search_value = str(search_value).lower()

    def matches(value):
        if isinstance(value, list):
            return any(
                search_value in str(item).lower()
                for item in value
            )

        return search_value in str(value).lower()

    return catalog[catalog[column_name].apply(matches)]
